-- 1. Enable Foreign Key support in SQLite 
PRAGMA foreign_keys = ON;


-- 2. Create the Users table
CREATE TABLE users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT NOT NULL UNIQUE,
    name TEXT,
    plan TEXT,
    password_hash TEXT NOT NULL,
    height INTEGER,
    weight INTEGER,
    fitness_goal TEXT,
    profile_picture TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP

);

-- 3. Create the pr table
CREATE TABLE pr (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    deadlift INTEGER,
    squat INTEGER,
    bench INTEGER,
    FOREIGN KEY (user_id) REFERENCES users(id)
);
-- 4.
CREATE TABLE nutrion (
    user_id INTEGER NOT NULL ,
    calories INTEGER,
    maintain INTEGER,
    protien INTEGER,
    carbs INTEGER,
    fats INTEGER,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 5. Create the WORKOUT table
  CREATE TABLE workout (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    workout_name TEXT NOT NULL,
    details TEXT,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 6. Completed workout sessions
CREATE TABLE workout_log (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    workout_name TEXT NOT NULL,
    notes TEXT,
    completed_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    session_date TEXT,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 7. Exercises recorded for each completed session
CREATE TABLE exercise_log (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    session_id INTEGER NOT NULL,
    exercise_name TEXT NOT NULL,
    sets INTEGER NOT NULL,
    reps INTEGER NOT NULL,
    weight_kg REAL NOT NULL,
    FOREIGN KEY (session_id) REFERENCES workout_log(id)
);

-- 8. Daily nutrition totals
CREATE TABLE daily_nutrition (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    log_date TEXT NOT NULL,
    calories REAL NOT NULL,
    protein REAL,
    carbs REAL,
    fats REAL,
    UNIQUE (user_id, log_date),
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 9. Body-composition measurements
CREATE TABLE body_measurement_log (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    measured_on TEXT NOT NULL,
    body_fat REAL,
    chest REAL,
    waist REAL,
    arms REAL,
    thighs REAL,
    UNIQUE (user_id, measured_on),
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 10. Progress photos
CREATE TABLE progress_photo (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    photo_date TEXT NOT NULL,
    filename TEXT NOT NULL,
    caption TEXT,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 11. Daily body-weight history
CREATE TABLE weight_log (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    weight REAL NOT NULL CHECK (weight > 0),
    measured_on TEXT NOT NULL,
    recorded_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (user_id, measured_on),
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 12. Dated personal-record improvements
CREATE TABLE pr_history (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    lift TEXT NOT NULL CHECK (lift IN ('bench', 'squat', 'deadlift')),
    weight_kg REAL NOT NULL,
    achieved_on TEXT NOT NULL,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 13. Paired before-and-after progress photos
CREATE TABLE photo_memorial (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    before_filename TEXT NOT NULL,
    before_date TEXT NOT NULL,
    after_filename TEXT,
    after_date TEXT,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 14. Food catalog used by nutrition logging
CREATE TABLE foods (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL COLLATE NOCASE UNIQUE,
    calories REAL NOT NULL CHECK (calories >= 0),
    protein REAL NOT NULL CHECK (protein >= 0),
    carbs REAL NOT NULL CHECK (carbs >= 0),
    fats REAL NOT NULL CHECK (fats >= 0)
);

-- 15. Foods recorded by each user
CREATE TABLE food_log (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    food_id INTEGER NOT NULL,
    food_name TEXT NOT NULL,
    grams REAL NOT NULL CHECK (grams > 0),
    calories REAL NOT NULL CHECK (calories >= 0),
    protein REAL NOT NULL CHECK (protein >= 0),
    carbs REAL NOT NULL CHECK (carbs >= 0),
    fats REAL NOT NULL CHECK (fats >= 0),
    logged_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (food_id) REFERENCES foods(id)
);

CREATE INDEX food_log_user_logged_at
    ON food_log (user_id, logged_at);
